\section*{Part 1: Queries}

\begin{enumerate}

%--------------------------------------------------
\item {\large Frugal doctors}

{~}\\
{\large
-- DINs that have a listed price. \\[5pt]
$
PricedDIN(\var{DIN}) :=
  \Pi_{\var{DIN}}(\var{Price})
$ \\[10pt]

-- Generic drugs together with their price. \\[5pt]
$
GenPrice(\var{DIN}, \var{brand}, \var{manufacturer}, \var{price}) \\
  := \Pi_{\var{DIN}, \var{brand}, \var{manufacturer}, \var{price}}
  (\var{Generic} \bowtie_{\var{Generic.DIN} = \var{Price.DIN}} \var{Price})
$ \\[10pt]

-- Generics that are not the cheapest for their brand. \\[5pt]
$
NonCheapest(\var{DIN}) :=
  \Pi_{G1.\var{DIN}}
  \sigma_{\substack{
    G1.\var{brand} = G2.\var{brand} \\
    \wedge \\
    G2.\var{price} < G1.\var{price}
  }}
  [ (\rho_{G1}GenPrice) \times (\rho_{G2}GenPrice) ]
$ \\[10pt]

-- Cheapest generics (ties allowed). \\[5pt]
$
CheapestGen(\var{DIN}) :=
  \Pi_{\var{DIN}}(GenPrice) - NonCheapest
$ \\[10pt]

-- Brand drugs that have no generic alternative and are priced. \\[5pt]
$
BrandNoGen(\var{DIN}) :=
  (\Pi_{\var{DIN}}(\var{Product}) \cap PricedDIN)
  -
  \Pi_{\var{brand}}(\var{Generic})
$ \\[10pt]

-- Allowed drugs are cheapest generics or brand drugs with no generics. \\[5pt]
$
AllowedDIN(\var{DIN}) :=
  CheapestGen \cup BrandNoGen
$ \\[10pt]

-- Prescriptions involving priced drugs. \\[5pt]
$
RxPriced :=
  \var{Prescription} \bowtie_{\var{drug} = \var{DIN}} PricedDIN
$ \\[10pt]

-- Doctors who prescribed a disallowed drug. \\[5pt]
$
BadDoctor(\var{doctor}) :=
  \Pi_{\var{doctor}}
  (\sigma_{\neg(\var{drug} = \var{DIN})}
   (RxPriced \times AllowedDIN))
$ \\[10pt]

-- Doctors who prescribed at least two different priced drugs. \\[5pt]
$
TwoDrugs(\var{doctor}) :=
  \Pi_{P1.\var{doctor}}
  \sigma_{\substack{
    P1.\var{doctor} = P2.\var{doctor} \\
    \wedge \\
    P1.\var{drug} \neq P2.\var{drug}
  }}
  [ (\rho_{P1}RxPriced) \times (\rho_{P2}RxPriced) ]
$ \\[10pt]

-- Final answer. \\[5pt]
$
Answer(\var{doctor}) :=
  TwoDrugs - BadDoctor
$
}

%--------------------------------------------------
\item {\large Potential doctor shopping}

{~}\\
{\large
-- Map each drug to its brand key. \\[5pt]
$
DrugKey(\var{drug}, \var{key}) :=
  \Pi_{\var{DIN}, \var{DIN}}(\var{Product})
  \cup
  \Pi_{\var{DIN}, \var{brand}}(\var{Generic})
$ \\[10pt]

-- Prescriptions with brand key attached. \\[5pt]
$
RxKey :=
  \var{Prescription} \bowtie_{\var{drug} = \var{DIN}} DrugKey
$ \\[10pt]

-- Patients with two prescriptions of same brand from different doctors. \\[5pt]
$
Shopper(\var{OHIP}) :=
  \Pi_{R1.\var{patient}}
  \sigma_{\substack{
    R1.\var{patient} = R2.\var{patient} \\
    \wedge \\
    R1.\var{key} = R2.\var{key} \\
    \wedge \\
    R1.\var{doctor} \neq R2.\var{doctor}
  }}
  [ (\rho_{R1}RxKey) \times (\rho_{R2}RxKey) ]
$ \\[10pt]

-- Final answer. \\[5pt]
$
Answer(\var{OHIP}, \var{name}, \var{phone}) :=
  \Pi_{\var{OHIP}, \var{name}, \var{phone}}
  (\var{Patient} \bowtie Shopper)
$
}

\item {\large Drug shortage}

{~}\\
{\large
-- Unfilled prescriptions. \\[5pt]
$
Unfilled :=
  \var{Prescription}
  -
  (\var{Prescription} \bowtie \var{Filled})
$ \\[10pt]

-- Drugs with at least three unfilled prescriptions. \\[5pt]
$
ThreeUnfilled(\var{DIN}) :=
  \Pi_{U1.\var{drug}}
  \sigma_{\substack{
    U1.\var{drug} = U2.\var{drug} \\
    \wedge \\
    U2.\var{drug} = U3.\var{drug} \\
    \wedge \\
    U1.\var{RxID} \neq U2.\var{RxID} \neq U3.\var{RxID}
  }}
  [ (\rho_{U1}Unfilled) \times (\rho_{U2}Unfilled) \times (\rho_{U3}Unfilled) ]
$ \\[10pt]

-- Drugs prescribed to at least two different patients. \\[5pt]
$
TwoPatients(\var{DIN}) :=
  \Pi_{U1.\var{drug}}
  \sigma_{\substack{
    U1.\var{drug} = U2.\var{drug} \\
    \wedge \\
    U1.\var{patient} \neq U2.\var{patient}
  }}
  [ (\rho_{U1}Unfilled) \times (\rho_{U2}Unfilled) ]
$ \\[10pt]

-- Shortage drugs. \\[5pt]
$
Shortage(\var{DIN}) :=
  ThreeUnfilled \cap TwoPatients
$ \\[10pt]

-- Manufacturer information. \\[5pt]
$
Answer(\var{DIN}, \var{manufacturer}) :=
  Shortage \bowtie
  (\Pi_{\var{DIN}, \var{manufacturer}}(\var{Product})
   \cup
   \Pi_{\var{DIN}, \var{manufacturer}}(\var{Generic}))
$
}

\item {\large Protecting drug patents}

{~}\\
{\large
-- Brand ingredients. \\[5pt]
$
BrandIng(\var{DIN}, \var{ingredient}) :=
  \Pi_{\var{DIN}, \var{ingredient}}(\var{Contains})
$ \\[10pt]

-- Pairs of distinct brand drugs. \\[5pt]
$
Pairs(\var{DIN1}, \var{DIN2}) :=
  \Pi_{P1.\var{DIN}, P2.\var{DIN}}
  \sigma_{P1.\var{DIN} < P2.\var{DIN}}
  [ (\rho_{P1}\var{Product}) \times (\rho_{P2}\var{Product}) ]
$ \\[10pt]

-- Ingredient mismatches imply protection. \\[5pt]
$
Answer :=
  Pairs
  -
  \Pi_{\var{DIN1}, \var{DIN2}}
  (BrandIng \cap BrandIng)
$
}

\item {\large Patients at risk}

{~}\\
{\large
-- Drug ingredients including generics. \\[5pt]
$
DrugIng :=
  \Pi_{\var{DIN}, \var{ingredient}}(\var{Contains})
  \cup
  \Pi_{G.\var{DIN}, C.\var{ingredient}}
  (\var{Generic} \bowtie_{\var{brand} = \var{DIN}} \var{Contains})
$ \\[10pt]

-- Interacting drug pairs. \\[5pt]
$
Interacting :=
  \Pi_{D1.\var{DIN}, D2.\var{DIN}}
  \sigma_{\substack{
    I.\var{ingredient1} = D1.\var{ingredient} \\
    \wedge \\
    I.\var{ingredient2} = D2.\var{ingredient}
  }}
  [ (\rho_{D1}DrugIng) \times (\rho_{D2}DrugIng) \times \var{Interaction} ]
$ \\[10pt]

-- Risk events. \\[5pt]
$
Answer(\var{doctor}, \var{date}) :=
  \Pi_{P1.\var{doctor}, P1.\var{date}}
  \sigma_{\substack{
    P1.\var{patient} = P2.\var{patient} \\
    \wedge \\
    P1.\var{doctor} = P2.\var{doctor} \\
    \wedge \\
    P1.\var{date} = P2.\var{date} \\
    \wedge \\
    P1.\var{drug} \neq P2.\var{drug}
  }}
  [ (\rho_{P1}\var{Prescription})
    \times
    (\rho_{P2}\var{Prescription})
    \times
    Interacting ]
$
}
